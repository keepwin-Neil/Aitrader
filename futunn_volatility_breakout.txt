{波动率突破策略 - Volatility Breakout}
{参数：K系数、EMA周期、最小波幅比}

K_FACTOR:=0.3;
EMA_N:=50;
MIN_RANGE:=0.3;

{昨日振幅}
PREV_RANGE:=REF(HIGH,1)-REF(LOW,1);

{突破线 = 今开 + 昨日振幅 × K}
UPPER:OPEN+PREV_RANGE*K_FACTOR,COLORF44336,LINETHICK1,DOTLINE;

{趋势线 EMA}
EMA_TREND:EMA(CLOSE,EMA_N),COLOR9C27B0,LINETHICK2;

{ATR(14)}
TR_VAL:=MAX(MAX(HIGH-LOW,ABS(HIGH-REF(CLOSE,1))),ABS(LOW-REF(CLOSE,1)));
ATR14:=MA(TR_VAL,14);

{波幅比 = 今日振幅 / ATR}
RNG_RATIO:=(HIGH-LOW)/ATR14;

{买入：收盘突破上轨 + 价格在EMA之上 + 波幅够大}
BUY_COND:=CLOSE>UPPER AND CLOSE>EMA_TREND AND RNG_RATIO>MIN_RANGE;

{卖出：跌破趋势线}
SELL_COND:=CLOSE<EMA_TREND;

DRAWICON(BUY_COND AND REF(BUY_COND,1)=0,LOW*0.99,1);
DRAWICON(SELL_COND AND REF(SELL_COND,1)=0,HIGH*1.01,2);
