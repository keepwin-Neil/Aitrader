N:=20;
M1:=3;
M2:=30;
TH:=0.3;

FAST_SC:=2.0/(M1+1);
SLOW_SC:=2.0/(M2+1);

DIRECTION:=ABS(CLOSE-REF(CLOSE,N));
VOLATILITY:=SUM(ABS(CLOSE-REF(CLOSE,1)),N);
ER:=IF(VOLATILITY>0,DIRECTION/VOLATILITY,0);

SC:=(ER*(FAST_SC-SLOW_SC)+SLOW_SC)*(ER*(FAST_SC-SLOW_SC)+SLOW_SC);

KAMA:DMA(CLOSE,SC),COLORFF9800,LINETHICK2;

KAMA_DIFF:=(KAMA-REF(KAMA,1))/REF(KAMA,1)*100;
KAMA_DIFF_MA:=MA(KAMA_DIFF,5);

BUY_COND:=KAMA_DIFF>TH AND KAMA_DIFF>KAMA_DIFF_MA AND CLOSE>KAMA;
SELL_COND:=KAMA_DIFF<0 OR CLOSE<KAMA;

DRAWICON(BUY_COND AND REF(BUY_COND,1)=0,LOW*0.99,1);
DRAWICON(SELL_COND AND REF(SELL_COND,1)=0,HIGH*1.01,2);
