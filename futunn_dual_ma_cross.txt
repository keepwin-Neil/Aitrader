{双均线交叉策略 - 金叉买入 死叉卖出}

FAST_N:=10;
SLOW_N:=25;

{===== 均线计算（SMA版本） =====}
MA_FAST:MA(CLOSE,FAST_N),COLORFF9800,LINETHICK2;
MA_SLOW:MA(CLOSE,SLOW_N),COLOR2196F3,LINETHICK2;

{===== 买卖信号 =====}
{金叉：快线上穿慢线}
BUY_COND:=CROSS(MA_FAST,MA_SLOW);
{死叉：快线下穿慢线}
SELL_COND:=CROSS(MA_SLOW,MA_FAST);

DRAWICON(BUY_COND,LOW*0.99,1);
DRAWICON(SELL_COND,HIGH*1.01,2);
